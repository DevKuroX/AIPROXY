### Step 3: Update Logout Function

```typescript
// Before
const logout = () => {
  localStorage.removeItem('token');
  setUser(null);
  router.push('/login');
};

// After
import { clearToken } from '@/lib/api';

const logout = () => {
  clearToken();
  setUser(null);
  router.push('/login');
};
```

### Step 4: Update userStore

```typescript
// src/store/userStore.ts
import { clearToken, getToken } from '@/lib/api';

export const useUserStore = create<UserStore>((set) => ({
  user: null,
  
  logout: () => {
    clearToken();
    set({ user: null });
  },
  
  isAuthenticated: () => {
    return !!getToken();
  }
}));
```
