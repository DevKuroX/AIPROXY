### Step 2: Identify Token-Related Fields

Look for:
- `token` field in state
- `setToken` action
- `clearToken` action

### Step 3: Refactor userStore

```typescript
// Before
interface UserStore {
  user: User | null;
  token: string | null;
  setUser: (user: User) => void;
  setToken: (token: string) => void;
  logout: () => void;
}

// After
import { getToken, setToken, clearToken } from '@/lib/api';

interface UserStore {
  user: User | null;
  // No token field
  setUser: (user: User) => void;
  logout: () => void;
  isAuthenticated: () => boolean;
}

export const useUserStore = create<UserStore>((set) => ({
  user: null,
  
  setUser: (user) => set({ user }),
  
  logout: () => {
    clearToken();
    set({ user: null });
  },
  
  isAuthenticated: () => !!getToken()
}));
```

### Step 4: Find userStore Token Consumers

```bash
rg -n "useUserStore.*token" src/
rg -n "\.token" src/store/
```

### Step 5: Update Consumers

```typescript
// Before
const { token, user } = useUserStore();

// After
import { getToken } from '@/lib/api';

const user = useUserStore((s) => s.user);
const token = getToken();
```

### Step 6: Verify Build

```bash
tsc --noEmit
npm run build
```
