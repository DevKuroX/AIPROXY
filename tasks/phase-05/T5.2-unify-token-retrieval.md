## Execution

### Step 1: Create getToken Function

In `src/lib/api.ts`:

```typescript
/**
 * Retrieves the authentication token from localStorage.
 * This is the ONLY place token should be retrieved.
 */
export function getToken(): string | null {
  if (typeof window === 'undefined') {
    return null;
  }
  return localStorage.getItem('token');
}

/**
 * Stores the authentication token in localStorage.
 */
export function setToken(token: string): void {
  localStorage.setItem('token', token);
}

/**
 * Removes the authentication token from localStorage.
 */
export function clearToken(): void {
  localStorage.removeItem('token');
}
```

### Step 3: Replace with Unified Functions

```typescript
// Before
const token = localStorage.getItem('token');
localStorage.setItem('token', newToken);
localStorage.removeItem('token');

// After
import { getToken, setToken, clearToken } from '@/lib/api';

const token = getToken();
setToken(newToken);
clearToken();
```

### Step 4: Update userStore

If userStore has token state:

```typescript
// Before
const useUserStore = create((set) => ({
  token: localStorage.getItem('token'),
  setToken: (t) => {
    localStorage.setItem('token', t);
    set({ token: t });
  }
}));

// After
import { getToken, setToken } from '@/lib/api';

const useUserStore = create((set) => ({
  // Token state removed - use getToken() directly
  isAuthenticated: () => !!getToken(),
  login: (token) => {
    setToken(token);
    set({ user: ... });
  },
  logout: () => {
    clearToken();
    set({ user: null });
  }
}));
```
