### Step 3: Create combos-api.ts

```typescript
// src/lib/combos-api.ts
import { authFetch } from './http';
import type { Combo } from '@/shared/contracts';

export interface CombosResponse {
  combos: Combo[];
}

export async function getCombos(): Promise<CombosResponse> {
  return authFetch<CombosResponse>('/api/combos');
}

export async function createCombo(combo: Partial<Combo>): Promise<Combo> {
  return authFetch<Combo>('/api/combos', {
    method: 'POST',
    body: JSON.stringify(combo)
  });
}

export async function updateCombo(id: string, combo: Partial<Combo>): Promise<Combo> {
  return authFetch<Combo>(`/api/combos/${id}`, {
    method: 'PUT',
    body: JSON.stringify(combo)
  });
}

export async function deleteCombo(id: string): Promise<void> {
  return authFetch<void>(`/api/combos/${id}`, {
    method: 'DELETE'
  });
}
```
