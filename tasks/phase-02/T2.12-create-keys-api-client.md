### Step 3: Create keys-api.ts

```typescript
// src/lib/keys-api.ts
import { authFetch } from './http';

export interface ApiKey {
  id: string;
  name: string;
  key: string;
  createdAt: string;
  lastUsed?: string;
}

export interface KeysResponse {
  keys: ApiKey[];
}

export async function getKeys(): Promise<KeysResponse> {
  return authFetch<KeysResponse>('/api/keys');
}

export async function createKey(name: string): Promise<ApiKey> {
  return authFetch<ApiKey>('/api/keys', {
    method: 'POST',
    body: JSON.stringify({ name })
  });
}

export async function deleteKey(id: string): Promise<void> {
  return authFetch<void>(`/api/keys/${id}`, {
    method: 'DELETE'
  });
}
```
