### Step 3: Create providers-api.ts

```typescript
// src/lib/providers-api.ts
import { authFetch } from './http';
import type { Provider } from '@/shared/contracts';

export interface ProvidersResponse {
  providers: Provider[];
}

export async function getProviders(): Promise<ProvidersResponse> {
  return authFetch<ProvidersResponse>('/api/providers');
}

export async function createProvider(provider: Partial<Provider>): Promise<Provider> {
  return authFetch<Provider>('/api/providers', {
    method: 'POST',
    body: JSON.stringify(provider)
  });
}

export async function updateProvider(id: string, provider: Partial<Provider>): Promise<Provider> {
  return authFetch<Provider>(`/api/providers/${id}`, {
    method: 'PUT',
    body: JSON.stringify(provider)
  });
}

export async function deleteProvider(id: string): Promise<void> {
  return authFetch<void>(`/api/providers/${id}`, {
    method: 'DELETE'
  });
}
```
