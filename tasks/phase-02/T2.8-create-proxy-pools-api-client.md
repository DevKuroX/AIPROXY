### Step 3: Create proxy-pools-api.ts

```typescript
// src/lib/proxy-pools-api.ts
import { authFetch } from './http';

export interface ProxyPool {
  id: string;
  name: string;
  proxies: string[];
  enabled: boolean;
}

export interface ProxyPoolsResponse {
  pools: ProxyPool[];
}

export async function getProxyPools(): Promise<ProxyPoolsResponse> {
  return authFetch<ProxyPoolsResponse>('/api/proxy-pools');
}

export async function createProxyPool(pool: Partial<ProxyPool>): Promise<ProxyPool> {
  return authFetch<ProxyPool>('/api/proxy-pools', {
    method: 'POST',
    body: JSON.stringify(pool)
  });
}

export async function updateProxyPool(id: string, pool: Partial<ProxyPool>): Promise<ProxyPool> {
  return authFetch<ProxyPool>(`/api/proxy-pools/${id}`, {
    method: 'PUT',
    body: JSON.stringify(pool)
  });
}

export async function deleteProxyPool(id: string): Promise<void> {
  return authFetch<void>(`/api/proxy-pools/${id}`, {
    method: 'DELETE'
  });
}
```
