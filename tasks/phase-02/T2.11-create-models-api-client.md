### Step 3: Create models-api.ts

```typescript
// src/lib/models-api.ts
import { authFetch } from './http';

export interface Model {
  id: string;
  name: string;
  provider: string;
  enabled: boolean;
}

export interface ModelsResponse {
  models: Model[];
}

export async function getModels(): Promise<ModelsResponse> {
  return authFetch<ModelsResponse>('/api/models');
}

export async function toggleModel(id: string, enabled: boolean): Promise<void> {
  return authFetch<void>(`/api/models/${id}/toggle`, {
    method: 'POST',
    body: JSON.stringify({ enabled })
  });
}
```
