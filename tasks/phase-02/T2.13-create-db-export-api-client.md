### Step 3: Create db-export-api.ts

```typescript
// src/lib/db-export-api.ts
import { authFetch } from './http';

export interface ExportResponse {
  data: string;
  timestamp: string;
}

export async function exportDatabase(): Promise<ExportResponse> {
  return authFetch<ExportResponse>('/api/db/export');
}

export async function importDatabase(data: string): Promise<void> {
  return authFetch<void>('/api/db/import', {
    method: 'POST',
    body: JSON.stringify({ data })
  });
}
```
