### Step 3: Create pricing-api.ts

```typescript
// src/lib/pricing-api.ts
import { authFetch } from './http';

export interface PricingTier {
  id: string;
  name: string;
  price: number;
  tokens: number;
}

export interface PricingResponse {
  tiers: PricingTier[];
}

export async function getPricing(): Promise<PricingResponse> {
  return authFetch<PricingResponse>('/api/pricing');
}

export async function updatePricing(tiers: PricingTier[]): Promise<void> {
  return authFetch<void>('/api/pricing', {
    method: 'POST',
    body: JSON.stringify({ tiers })
  });
}
```
