### Step 2: Identify authFetch Patterns

For each file, identify:
- Token retrieval logic
- Header construction
- Error handling
- Fetch patterns

### Step 3: Import authFetch

```typescript
// Before
// Inline fetch logic with token...

// After
import { authFetch } from './http';
```

### Step 4: Replace Inline Fetch Logic

Replace all inline authenticated fetch calls with `authFetch`:

```typescript
// Before
const token = localStorage.getItem('token');
const response = await fetch('/api/analytics', {
  headers: { Authorization: `Bearer ${token}` }
});

// After
const response = await authFetch<AnalyticsResponse>('/api/analytics');
```

### Step 5: Update analytics-api.ts

```typescript
import { authFetch } from './http';

export async function getAnalytics() {
  return authFetch<AnalyticsResponse>('/api/analytics');
}

export async function getUsageStats() {
  return authFetch<UsageStatsResponse>('/api/analytics/usage');
}
```

### Step 6: Update oauth-api.ts

```typescript
import { authFetch } from './http';

export async function getOAuthProviders() {
  return authFetch<OAuthProvidersResponse>('/api/oauth/providers');
}
```
