### Step 2: Implement Client

```typescript
// src/lib/usage-api.ts

import { authFetch } from './http';
import type { UsageStats } from '@/shared/contracts';

export interface UsageHistory {
  // Match backend response shape
  date: string;
  requests: number;
  tokens: number;
}

export interface ChartData {
  labels: string[];
  datasets: {
    label: string;
    data: number[];
  }[];
}

/**
 * Get usage history for date range
 */
export async function getUsageHistory(
  startDate?: Date,
  endDate?: Date
): Promise<UsageHistory[]> {
  const params = new URLSearchParams();
  if (startDate) params.set('start', startDate.toISOString());
  if (endDate) params.set('end', endDate.toISOString());
  
  return authFetch<UsageHistory[]>(`/api/admin/usage/history?${params}`);
}

/**
 * Get usage statistics
 */
export async function getUsageStats(): Promise<UsageStats> {
  return authFetch<UsageStats>('/api/admin/usage/stats');
}

/**
 * Get chart data for usage visualization
 */
export async function getChartData(
  period: 'day' | 'week' | 'month' = 'week'
): Promise<ChartData> {
  return authFetch<ChartData>(`/api/admin/usage/chart?period=${period}`);
}

/**
 * Get active/pending requests
 */
export async function getActiveRequests(): Promise<unknown[]> {
  return authFetch<unknown[]>('/api/admin/usage/active');
}

/**
 * Get recent request logs
 */
export async function getRecentLogs(limit: number = 100): Promise<unknown[]> {
  return authFetch<unknown[]>(`/api/admin/usage/logs?limit=${limit}`);
}

/**
 * Get request details by ID
 */
export async function getRequestDetailById(id: string): Promise<unknown> {
  return authFetch<unknown>(`/api/admin/usage/requests/${id}`);
}
```
