### Step 1: List HAR Fixtures

```bash
cd /home/ubuntu/ai_proxy
ls -la fixtures/before/*.har
```

### Step 2: Create HAR Replay Script

```bash
cat > scripts/har-replay.sh << 'EOF'
#!/bin/bash

ENDPOINTS=(
  "/api/auth/me"
  "/api/health"
  "/api/version"
  "/api/providers"
  "/api/settings"
  "/api/usage"
  "/api/aliases"
  "/api/combos"
  # Add other endpoints
)

TOKEN="your-test-token"

for endpoint in "${ENDPOINTS[@]}"; do
  echo "Testing: $endpoint"
  
  # Compare with fixture if exists
  fixture="fixtures/before/${endpoint//\//-}.json"
  
  if [ -f "$fixture" ]; then
    response=$(curl -s -H "Authorization: Bearer $TOKEN" "http://localhost:3000$endpoint")
    expected=$(cat "$fixture")
    
    if [ "$response" == "$expected" ]; then
      echo "✓ PASS: $endpoint"
    else
      echo "✗ FAIL: $endpoint"
      echo "Expected: $expected"
      echo "Got: $response"
    fi
  fi
done
EOF

chmod +x scripts/har-replay.sh
```

### Step 3: Start Frontend

```bash
cd /home/ubuntu/ai_proxy/frontend
npm run dev
```

### Step 4: Run HAR Replay

```bash
./scripts/har-replay.sh
```
