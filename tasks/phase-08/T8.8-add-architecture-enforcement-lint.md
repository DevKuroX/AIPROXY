### Step 2: Add ESLint Rule

In `eslint.config.mjs` or `eslintrc.js`:

```javascript
// eslint.config.mjs
export default [
  {
    rules: {
      'no-restricted-imports': ['error', {
        patterns: [{
          group: ['@/lib/db/*'],
          message: 'SQLite is not allowed in frontend. Use backend API instead.'
        }, {
          group: ['@/lib/usageDb'],
          message: 'usageDb is deprecated. Use usage-api.ts instead.'
        }, {
          group: ['@/lib/localDb'],
          message: 'localDb is deprecated. Use appropriate API client instead.'
        }, {
          group: ['@/lib/dataDir'],
          message: 'dataDir is deprecated. Frontend has no filesystem access.'
        }, {
          group: ['@/lib/disabledModelsDb'],
          message: 'disabledModelsDb is deprecated. Use backend API.'
        }, {
          group: ['@/lib/requestDetailsDb'],
          message: 'requestDetailsDb is deprecated. Use backend API.'
        }, {
          group: ['@/lib/mitmAliasCache'],
          message: 'mitmAliasCache is deprecated.'
        }, {
          group: ['better-sqlite3'],
          message: 'SQLite is not allowed in frontend.'
        }, {
          group: ['sqlite3'],
          message: 'SQLite is not allowed in frontend.'
        }, {
          group: ['fs/promises'],
          message: 'Filesystem access not allowed in client-side code.'
        }]
      }]
    }
  }
];
```
