## Test Cases

1. **Method Forwarding**
   - GET request forwarded correctly
   - POST request forwarded with body
   - PUT/DELETE requests forwarded

2. **Header Forwarding**
   - Authorization header forwarded
   - Cookie header forwarded
   - Content-Type forwarded

3. **Streaming Passthrough**
   - SSE stream piped correctly
   - Binary stream piped correctly
   - Chunk ordering preserved

4. **Status Code Forwarding**
   - 200 response forwarded
   - 400/500 errors forwarded
   - 401/403 auth errors forwarded

5. **Error Passthrough**
   - Backend error body forwarded verbatim
   - Network error handled

```typescript
// src/lib/proxy.test.ts

import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import { NextRequest } from 'next/server';
import { proxyToBackend } from './proxy';

// Mock fetch
const mockFetch = vi.fn();
global.fetch = mockFetch;

describe('proxyToBackend', () => {
  beforeEach(() => {
    mockFetch.mockReset();
  });

  describe('method forwarding', () => {
    it('forwards GET request', async () => {
      mockFetch.mockResolvedValueOnce({
        status: 200,
        headers: new Headers({ 'content-type': 'application/json' }),
        text: () => Promise.resolve('{"ok":true}'),
      });

      const req = new NextRequest('http://localhost/api/test');
      const res = await proxyToBackend(req);

      expect(mockFetch).toHaveBeenCalledWith(
        expect.stringContaining('/api/test'),
        expect.objectContaining({ method: 'GET' })
      );
      expect(res.status).toBe(200);
    });

    it('forwards POST request with body', async () => {
      mockFetch.mockResolvedValueOnce({
        status: 201,
        headers: new Headers({ 'content-type': 'application/json' }),
        text: () => Promise.resolve('{"created":true}'),
      });

      const req = new NextRequest('http://localhost/api/test', {
        method: 'POST',
        body: JSON.stringify({ foo: 'bar' }),
        headers: { 'content-type': 'application/json' },
      });
      const res = await proxyToBackend(req);

      expect(mockFetch).toHaveBeenCalledWith(
        expect.any(String),
        expect.objectContaining({
          method: 'POST',
          body: expect.any(String),
        })
      );
    });
  });

  describe('header forwarding', () => {
    it('forwards Authorization header', async () => {
      mockFetch.mockResolvedValueOnce({
        status: 200,
        headers: new Headers(),
        text: () => Promise.resolve('{}'),
      });

      const req = new NextRequest('http://localhost/api/test', {
        headers: { authorization: 'Bearer token123' },
      });
      await proxyToBackend(req);

      expect(mockFetch).toHaveBeenCalledWith(
        expect.any(String),
        expect.objectContaining({
          headers: expect.any(Headers),
        })
      );
    });
  });

  describe('streaming', () => {
    it('pipes SSE stream', async () => {
      const streamBody = new ReadableStream({
        start(controller) {
          controller.enqueue('data: {"delta":"hello"}\n\n');
          controller.enqueue('data: [DONE]\n\n');
          controller.close();
        },
      });

      mockFetch.mockResolvedValueOnce({
        status: 200,
        headers: new Headers({ 'content-type': 'text/event-stream' }),
        body: streamBody,
      });

      const req = new NextRequest('http://localhost/api/chat');
      const res = await proxyToBackend(req);

      expect(res.body).toBeInstanceOf(ReadableStream);
    });
  });

  describe('status forwarding', () => {
    it('forwards 500 error', async () => {
      mockFetch.mockResolvedValueOnce({
        status: 500,
        headers: new Headers({ 'content-type': 'application/json' }),
        text: () => Promise.resolve('{"error":"Internal Server Error"}'),
      });

      const req = new NextRequest('http://localhost/api/test');
      const res = await proxyToBackend(req);

      expect(res.status).toBe(500);
    });
  });
});
```
