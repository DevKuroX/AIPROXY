## Behavior Requirements

1. Read `BACKEND_INTERNAL_URL` from server environment
2. Forward: method, path, query, body
3. Forward headers: `Authorization`, `Cookie`
4. Stream response body if `content-type` is:
   - `text/event-stream` (SSE)
   - `application/octet-stream` (binary)
5. Return backend status verbatim
6. Never parse business payloads

### Step 2: Implement proxyToBackend

```typescript
// src/lib/proxy.ts

import { NextRequest, NextResponse } from 'next/server';

const BACKEND_URL = process.env.BACKEND_INTERNAL_URL || 'http://localhost:20128';

/**
 * Thin proxy to backend.
 * 
 * - Forwards all request properties
 * - Streams response for SSE and binary
 * - Returns backend status verbatim
 */
export async function proxyToBackend(
  req: NextRequest,
  path?: string
): Promise<NextResponse> {
  // Determine target path
  const targetPath = path || req.nextUrl.pathname;
  const searchParams = req.nextUrl.search;
  const backendUrl = `${BACKEND_URL}${targetPath}${searchParams}`;
  
  // Forward headers
  const headers = new Headers();
  
  // Forward Authorization header
  const authHeader = req.headers.get('authorization');
  if (authHeader) {
    headers.set('Authorization', authHeader);
  }
  
  // Forward Cookie header
  const cookieHeader = req.headers.get('cookie');
  if (cookieHeader) {
    headers.set('Cookie', cookieHeader);
  }
  
  // Forward other relevant headers
  const contentType = req.headers.get('content-type');
  if (contentType) {
    headers.set('Content-Type', contentType);
  }
  
  // Build fetch options
  const fetchOptions: RequestInit = {
    method: req.method,
    headers,
  };
  
  // Forward body for non-GET requests
  if (req.method !== 'GET' && req.method !== 'HEAD') {
    fetchOptions.body = await req.text();
  }
  
  // Execute backend request
  const backendResponse = await fetch(backendUrl, fetchOptions);
  
  // Determine if streaming is needed
  const responseContentType = backendResponse.headers.get('content-type') || '';
  const shouldStream = 
    responseContentType.includes('text/event-stream') ||
    responseContentType.includes('application/octet-stream');
  
  // Build response headers
  const responseHeaders = new Headers();
  
  // Forward relevant response headers
  if (responseContentType) {
    responseHeaders.set('Content-Type', responseContentType);
  }
  
  // Forward set-cookie if present
  const setCookie = backendResponse.headers.get('set-cookie');
  if (setCookie) {
    responseHeaders.set('Set-Cookie', setCookie);
  }
  
  if (shouldStream) {
    // Stream response body
    return new NextResponse(backendResponse.body, {
      status: backendResponse.status,
      headers: responseHeaders,
    });
  } else {
    // Buffer and return
    const body = await backendResponse.text();
    return new NextResponse(body, {
      status: backendResponse.status,
      headers: responseHeaders,
    });
  }
}
```
