### Step 1: Create SSE Consumer Module

```typescript
// src/sse/consumer.ts

export type StreamEventType = 'delta' | 'done' | 'error' | 'retry' | 'usage';

export interface StreamEvent {
  type: StreamEventType;
  content?: string;
  message?: string;
  code?: string;
  retry?: number;
  usage?: {
    promptTokens: number;
    completionTokens: number;
    totalTokens: number;
  };
}

export interface StreamCallbacks {
  onDelta: (content: string) => void;
  onDone: () => void;
  onError: (message: string, code?: string) => void;
  onRetry?: (retryAfter: number) => void;
  onUsage?: (usage: StreamEvent['usage']) => void;
}

/**
 * Consume POST-SSE stream (for chat endpoints)
 */
export async function consumePostStream(
  response: Response,
  callbacks: StreamCallbacks
): Promise<void> {
  const reader = response.body?.getReader();
  if (!reader) {
    throw new Error('No response body');
  }

  const decoder = new TextDecoder();
  let buffer = '';

  try {
    while (true) {
      const { done, value } = await reader.read();
      
      if (done) break;

      buffer += decoder.decode(value, { stream: true });
      
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';

      for (const line of lines) {
        if (line.startsWith('data: ')) {
          const data = line.slice(6);
          if (data === '[DONE]') {
            callbacks.onDone();
            return;
          }
          
          try {
            const event: StreamEvent = JSON.parse(data);
            handleEvent(event, callbacks);
          } catch {
            // Ignore parse errors
          }
        }
      }
    }
  } finally {
    reader.releaseLock();
  }
}

/**
 * Consume GET-SSE stream (for event streams)
 */
export function consumeGetStream(
  url: string,
  callbacks: StreamCallbacks
): EventSource {
  const eventSource = new EventSource(url);
  
  eventSource.onmessage = (event) => {
    try {
      const data: StreamEvent = JSON.parse(event.data);
      handleEvent(data, callbacks);
    } catch {
      // Ignore parse errors
    }
  };
  
  eventSource.onerror = () => {
    callbacks.onError('Connection lost');
  };
  
  return eventSource;
}

function handleEvent(event: StreamEvent, callbacks: StreamCallbacks): void {
  switch (event.type) {
    case 'delta':
      if (event.content) {
        callbacks.onDelta(event.content);
      }
      break;
    case 'done':
      callbacks.onDone();
      break;
    case 'error':
      callbacks.onError(event.message || 'Unknown error', event.code);
      break;
    case 'retry':
      if (event.retry && callbacks.onRetry) {
        callbacks.onRetry(event.retry);
      }
      break;
    case 'usage':
      if (event.usage && callbacks.onUsage) {
        callbacks.onUsage(event.usage);
      }
      break;
  }
}
```
