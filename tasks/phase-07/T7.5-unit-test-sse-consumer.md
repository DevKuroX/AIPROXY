```typescript
// src/sse/consumer.test.ts
import { consumePostStream, type StreamCallbacks } from './consumer';

describe('consumePostStream', () => {
  const createMockResponse = (chunks: string[]): Response => {
    const encoder = new TextEncoder();
    const stream = new ReadableStream({
      start(controller) {
        for (const chunk of chunks) {
          controller.enqueue(encoder.encode(chunk));
        }
        controller.close();
      }
    });
    
    return {
      body: stream,
    } as Response;
  };

  it('processes chunks in order', async () => {
    const deltas: string[] = [];
    const callbacks: StreamCallbacks = {
      onDelta: (content) => deltas.push(content),
      onDone: () => {},
      onError: () => {},
    };

    const response = createMockResponse([
      'data: {"type":"delta","content":"first"}\n',
      'data: {"type":"delta","content":"second"}\n',
      'data: {"type":"done"}\n',
    ]);

    await consumePostStream(response, callbacks);

    expect(deltas).toEqual(['first', 'second']);
  });

  it('handles done event', async () => {
    let doneCalled = false;
    const callbacks: StreamCallbacks = {
      onDelta: () => {},
      onDone: () => { doneCalled = true; },
      onError: () => {},
    };

    const response = createMockResponse([
      'data: {"type":"done"}\n',
    ]);

    await consumePostStream(response, callbacks);

    expect(doneCalled).toBe(true);
  });

  it('handles error event', async () => {
    let errorMessage = '';
    const callbacks: StreamCallbacks = {
      onDelta: () => {},
      onDone: () => {},
      onError: (msg) => { errorMessage = msg; },
    };

    const response = createMockResponse([
      'data: {"type":"error","message":"test error"}\n',
    ]);

    await consumePostStream(response, callbacks);

    expect(errorMessage).toBe('test error');
  });

  it('handles abort', async () => {
    const callbacks: StreamCallbacks = {
      onDelta: () => {},
      onDone: () => {},
      onError: () => {},
    };

    const controller = new AbortController();
    const response = createMockResponse([]);

    // Abort immediately
    controller.abort();

    // Should not throw
    await consumePostStream(response, callbacks);
  });
});
```
