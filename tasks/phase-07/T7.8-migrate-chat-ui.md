### Step 3: Update to Use New Consumer

```typescript
// Before
import { parseStream } from '@/open-sse/parser';

const handleStream = async (response: Response) => {
  for await (const chunk of parseStream(response)) {
    if (chunk.finish_reason) { ... }
    setContent(prev => prev + chunk.delta);
  }
};

// After
import { consumePostStream } from '@/sse/consumer';

const handleStream = async (response: Response) => {
  await consumePostStream(response, {
    onDelta: (content) => {
      setContent(prev => prev + content);
    },
    onDone: () => {
      setIsStreaming(false);
    },
    onError: (message) => {
      setError(message);
      setIsStreaming(false);
    },
    onRetry: (retryAfter) => {
      setShowRetryIndicator(true);
    },
  });
};
```
