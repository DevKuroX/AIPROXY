## Test Matrix

### 1. Short Response

- **Test**: Send a chat request expecting < 100 tokens
- **Verify**: Response completes successfully
- **Pass criteria**: Full response received, `done` event fired

### 2. Long Response

- **Test**: Send a chat request expecting > 1000 tokens
- **Verify**: Stream continues, chunks arrive in order
- **Pass criteria**: All chunks received in order, no drops

### 3. Abort Mid-Stream

- **Test**: Start streaming, abort after 3 chunks
- **Verify**: Connection closes cleanly, backend notified
- **Pass criteria**: No hanging connections, proper cleanup

### 4. Network Blip

- **Test**: Simulate network interruption during stream
- **Verify**: Error propagates to client
- **Pass criteria**: Error event received, client can retry

### 5. Provider 5xx

- **Test**: Force backend to return 500 error mid-stream
- **Verify**: Error propagates to client correctly
- **Pass criteria**: Error event with correct message

### 6. Provider Auth Fail

- **Test**: Send request with invalid/expired provider token
- **Verify**: Auth error propagates correctly
- **Pass criteria**: 401/403 error returned to client

// tests/streaming/matrix.test.ts

import { describe, it, expect } from 'vitest';

const BACKEND_URL = process.env.BACKEND_URL || 'http://localhost:20128';

describe('Streaming Matrix', () => {
  describe('Short Response', () => {
    it('completes short chat', async () => {
      const response = await fetch(`${BACKEND_URL}/v1/chat/completions`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: 'gpt-4',
          messages: [{ role: 'user', content: 'Say "hello"' }],
          stream: true,
        }),
      });

      expect(response.status).toBe(200);
      expect(response.headers.get('content-type')).toContain('text/event-stream');
      
      // Consume stream
      const reader = response.body?.getReader();
      let done = false;
      while (!done) {
        const result = await reader?.read();
        done = result?.done || false;
      }
      
      expect(done).toBe(true);
    });
  });

  describe('Long Response', () => {
    it('streams long response in order', async () => {
      // Test with prompt expecting long response
    });
  });

  describe('Abort Mid-Stream', () => {
    it('handles abort cleanly', async () => {
      const controller = new AbortController();
      
      const response = await fetch(`${BACKEND_URL}/v1/chat/completions`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          model: 'gpt-4',
          messages: [{ role: 'user', content: 'Write a long story' }],
          stream: true,
        }),
        signal: controller.signal,
      });

      // Read 3 chunks, then abort
      const reader = response.body?.getReader();
      for (let i = 0; i < 3; i++) {
        await reader?.read();
      }
      
      controller.abort();
      
      // Verify cleanup
    });
  });

  // ... other test cases
});
